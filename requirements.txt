# The SUMO toolchain itself. Installing it from PyPI rather than Homebrew
# means `pip install -r requirements.txt` gets netconvert, sumo, duarouter and
# sumo-gui into the venv with no tap to trust and no XQuartz needed for the
# command-line tools.
eclipse-sumo>=1.27

httpx>=0.27
lxml>=5.0
pytest>=8.0

# NMDOT's crash counts reach the public as a PDF of bar charts, one per
# community per year, and the numbers wanted are the data labels printed over
# the bars. Reading them needs word positions, not page text: the labels are
# ordered by where they sit, not by where they appear in the file.
pdfplumber>=0.11

# from the eclipse-sumo binaries, and needs pyproj for the lat/lon conversion
# the network's geo-projection describes.
sumolib>=1.27
pyproj>=3.6
# Spatial index for nearest-edge lookup. Optional -- sumolib falls back to a
